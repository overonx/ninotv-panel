function doGet(e) { try { var action = e.parameter.action; var query = e.parameter.q; var ss = SpreadsheetApp.getActiveSpreadsheet(); // Mapeo de pestañas y nombres var sheetsToSearch = [ { key: "xuper", name: "Clientes", service: "Xúper TV" }, { key: "lion", name: "LIONTV", service: "Lion TV" }, { key: "red", name: "REDTV", service: "Red TV" }, { key: "flujo", name: "FLUJOTV", service: "Flujo TV" } ]; // Acción 1: Traer las métricas/contadores de las tarjetas if (action === 'stats') { var statsResult = {}; sheetsToSearch.forEach(function(item) { var sheet = ss.getSheetByName(item.name); var total = 0, activos = 0, vencer = 0, vencidos = 0, inactivos = 0; if (sheet) { var data = sheet.getDataRange().getValues(); // Ignoramos la fila 1 de encabezados for (var i = 1; i < data.length; i++) { var row = data[i]; if (row[0] && String(row[0]).trim() !== '') { total++; var estado = String(row[3] || '').toUpperCase().trim(); if (estado.indexOf('ACTIVO') !== -1) { activos++; } else if (estado.indexOf('POR VENC') !== -1 || estado.indexOf('VENCER') !== -1) { vencer++; } else if (estado.indexOf('VENCIDO') !== -1) { vencidos++; } else if (estado.indexOf('INACTIVO') !== -1) { inactivos++; } else { activos++; // Por defecto si tiene datos } } } } statsResult[item.key] = { total: total, activos: activos, vencer: vencer, vencidos: vencidos, inactivos: inactivos }; }); return ContentService.createTextOutput(JSON.stringify({ status: 'success', stats: statsResult })).setMimeType(ContentService.MimeType.JSON); } // Acción 2: Buscar cliente por nombre if (action === 'search' && query) { var results = []; query = query.toLowerCase().trim(); sheetsToSearch.forEach(function(item) { var sheet = ss.getSheetByName(item.name); if (sheet) { var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var row = data[i]; var nombre = String(row[0] || ''); if (nombre.toLowerCase().indexOf(query) !== -1) { var fechaVenc = 'N/A'; if (row[2]) { if (row[2] instanceof Date) { fechaVenc = Utilities.formatDate(row[2], Session.getScriptTimeZone(), "dd/MM/yyyy"); } else { fechaVenc = String(row[2]); } } results.push({ nombre: nombre, vencimiento: fechaVenc, estado: row[3] || 'ACTIVO', servicio: item.service }); } } } }); return ContentService.createTextOutput(JSON.stringify({ status: 'success', results: results })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Acción no válida' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { return doGet(e); }
